"""Independently derive committed golden samples from native fixture bytes.

Uses no energy_platform code, manifest mapping, runtime parser, or golden runner.
Validates positive fixtures only; quarantine behavior belongs to the platform tests.
Run from the repository root with PYTHONDONTWRITEBYTECODE=1 .venv/bin/python <this file>.
"""
from datetime import UTC, date, datetime, time, timedelta
from decimal import Decimal
from io import BytesIO
import json
from pathlib import Path
from zoneinfo import ZoneInfo
import xml.etree.ElementTree as ET

import openpyxl
import yaml

ZONE = ZoneInfo("Europe/Prague")
OTE_METRICS = {
    "ote_intraday_market": {"price_vwap": "Price", "volume_total": "Volume"},
    "ote_dam": {"price": "Price", "price_hourly": "HourlyPrice",
                "volume_total": "VolumeTotal", "emergency_state": "EmergencyState"},
}
SETTLEMENT = {"system_imbalance": "SystemImbalance", "imbalance_price": "SettlImbalancePrice",
              "counter_imbalance_price": "SettlCounterImbalancePrice"}
XLSX_METRICS = {
    "price_vwap": "Average price (EUR/MWh)",
    "volume_total": "Traded volume (MWh)",
    "volume_buy": "Traded volume - purchase (MWh)",
    "volume_sell": "Traded volume - sold (MWh)",
    "price_min": "Minimal price (EUR/MWh)",
    "price_max": "Maximal price (EUR/MWh)",
    "price_last": "Last price (EUR/MWh)",
}


def number(value):
    return None if value is None or str(value).strip() == "" else Decimal(str(value))


def instant(value):
    return (value if isinstance(value, datetime) else datetime.fromisoformat(value)).astimezone(UTC)


def indexed(day, index, resolution):
    minutes = {"PT15M": 15, "PT60M": 60}[resolution]
    start = datetime.combine(day, time(), ZONE).astimezone(UTC)
    finish = datetime.combine(day + timedelta(days=1), time(), ZONE).astimezone(UTC)
    count = int((finish - start).total_seconds() / (60 * minutes))
    assert 1 <= index <= count
    return start + timedelta(minutes=(index - 1) * minutes)


def from_xml(target, blob):
    rows = []
    root = ET.fromstring(blob)
    for element in root.iter():
        if target == "ceps_load":
            if element.tag.rsplit("}", 1)[-1] != "item" or "date" not in element.attrib:
                continue
            for metric, name in {"load_incl_pumping": "value1", "load": "value2"}.items():
                rows.append({"delivery_start_utc": instant(element.attrib["date"]),
                             "resolution": "PT15M", "metric": metric,
                             "value": number(element.attrib.get(name)), "source_version": "RT"})
        else:
            fields = {c.tag.rsplit("}", 1)[-1]: c.text for c in element}
            if not {"Date", "PeriodIndex", "PeriodResolution"} <= fields.keys():
                continue
            day = date.fromisoformat(fields["Date"][:10])
            index, resolution = int(fields["PeriodIndex"]), fields["PeriodResolution"]
            metrics = SETTLEMENT if target.startswith("ote_imbalance_settlement") else OTE_METRICS[target]
            for metric, field in metrics.items():
                rows.append({"delivery_start_utc": indexed(day, index, resolution),
                             "resolution": resolution, "period_index": index,
                             "metric": metric, "value": number(fields.get(field)),
                             "source_version": fields.get("Version")})
    return rows


def from_xlsx(blob, entry):
    workbook = openpyxl.load_workbook(BytesIO(blob), data_only=True, read_only=True)
    table = list(workbook.active.values)
    header = [" ".join(str(v).split()) if v is not None else "" for v in table[5]]
    day = instant(entry["scheduled_for"]).astimezone(ZONE).date()
    rows = []
    for record in table[6:]:
        if not isinstance(record[0], (int, float)):
            break
        index = int(record[0])
        for metric, label in XLSX_METRICS.items():
            rows.append({"delivery_start_utc": indexed(day, index, "PT15M"),
                         "resolution": "PT15M", "period_index": index, "metric": metric,
                         "value": number(record[header.index(label)]), "source_version": None})
    workbook.close()
    return rows


fixtures = samples = 0
native_rows = 0
summary = []
for target in sorted(Path("targets").iterdir()):
    if not (target / "tests/golden").is_dir():
        continue
    target_fixtures = target_samples = target_rows = 0
    for golden in sorted((target / "tests/golden").glob("*.yaml")):
        spec = yaml.safe_load(golden.read_text())
        expected = spec["expect"]
        if "quarantine" in expected:
            continue
        folder = target / "fixtures" / spec["fixture"]
        blob = (folder / "blob").read_bytes()
        entry = json.loads((folder / "entry.json").read_text())
        rows = from_xlsx(blob, entry) if target.name.endswith("_xlsx") else from_xml(target.name, blob)
        assert len(rows) == expected["row_count"], (str(golden), len(rows), expected["row_count"])
        for expected_row in expected.get("rows", []):
            ts = instant(expected_row["delivery_start_utc"])
            candidates = [r for r in rows if r["delivery_start_utc"] == ts
                          and r["resolution"] == expected_row["resolution"]
                          and r["metric"] == expected_row["metric"]
                          and ("source_version" not in expected_row
                               or r["source_version"] == str(expected_row["source_version"]))]
            if "period_index" in expected_row:
                candidates = [r for r in candidates if r["period_index"] == expected_row["period_index"]]
            assert len(candidates) == 1, (str(golden), expected_row, candidates)
            assert candidates[0]["value"] == number(expected_row["value"]), (str(golden), expected_row, candidates)
            target_samples += 1
        target_fixtures += 1
        target_rows += len(rows)
    fixtures += target_fixtures
    samples += target_samples
    native_rows += target_rows
    summary.append({"target": target.name, "positive_fixtures": target_fixtures,
                    "native_row_counts_checked": target_rows, "golden_values_checked": target_samples})
print(json.dumps({"status": "PASS", "energy_platform_imports": 0,
                  "positive_fixtures": fixtures, "native_rows_counted": native_rows,
                  "sample_values_and_utc_instants_checked": samples, "targets": summary}, indent=2))
